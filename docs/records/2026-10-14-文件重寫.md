## 可重跑的文件檢查

在專案根目錄，以 Python 3 執行以下程式。它只讀取檔案與 Git 狀態，不下載資料。相對連結與 Markdown 章節錨點會檢查，外部網址不連線驗證。歷史內容比對允許本次已說明的連結轉文字與附註，忽略檔尾空白差異。

```python
from pathlib import Path
from urllib.parse import unquote
import json
import re
import subprocess
import unicodedata

root = Path.cwd().resolve()
docs = [Path("README.md"), *sorted(Path("docs").rglob("*.md"))]
errors = []
links = 0

def headings(text):
    result = set()
    counts = {}
    fenced = False
    for line in text.splitlines():
        if line.startswith(chr(96) * 3):
            fenced = not fenced
        if fenced or not re.match(r"^#{1,6} ", line):
            continue
        title = re.sub(r"^#+ ", "", line).strip().lower()
        slug = "".join(c for c in title if c in "-_ " or unicodedata.category(c)[0] in "LN").replace(" ", "-")
        count = counts.get(slug, 0)
        counts[slug] = count + 1
        result.add(slug if count == 0 else slug + "-" + str(count))
    return result

for path in docs:
    text = path.read_text(encoding="utf-8")
    fenced = False
    for no, line in enumerate(text.splitlines(), 1):
        if line.startswith(chr(96) * 3):
            fenced = not fenced
        if fenced:
            continue
        for label, target in re.findall(r"\[([^\]\n]+)\]\(([^)\s]+)\)", line):
            if re.match(r"^[a-zA-Z][a-zA-Z0-9+.-]*:", target):
                continue
            links += 1
            location, _, anchor = unquote(target).partition("#")
            dest = (path.parent / location) if location else path
            if not dest.exists():
                errors.append(f"{path}:{no}: missing {target}")
            elif anchor and dest.suffix == ".md" and anchor not in headings(dest.read_text(encoding="utf-8")):
                errors.append(f"{path}:{no}: missing anchor {target}")

current = [Path("README.md"), *Path("docs").glob("*.md"),
           Path("docs/records/2026-10-14-文件重寫.md")]
for path in current:
    text = path.read_text(encoding="utf-8")
    if len(text.splitlines()) > 550:
        errors.append(f"{path}: over 550 lines")
    if text != text.rstrip() + "\n" or "\n\n\n" in text:
        errors.append(f"{path}: blank lines or final newline")
    if any(line.rstrip() != line or "\t" in line for line in text.splitlines()):
        errors.append(f"{path}: trailing whitespace or tabs")

def original(path):
    return subprocess.check_output(["git", "show", "HEAD:" + path.as_posix()]).decode("utf-8")

note = "\n\n## 2026-10-14 連結維護註記\n"
history_checked = 0
for folder in ("docs/records", "docs/archive"):
    tracked = subprocess.check_output(["git", "ls-files", "-z", folder]).decode("utf-8").split("\0")
    for name in filter(None, tracked):
        path = Path(name)
        before = original(path)
        after = path.read_text(encoding="utf-8")
        if folder == "docs/records":
            before = re.sub(r"\[([^\]]+)\]\((\.\./plans/[^)]+)\)",
                            lambda m: m[1] + " " + chr(96) + m[2] + chr(96), before)
            after = after.split(note)[0]
        if before.rstrip() != after.rstrip():
            errors.append(f"{path}: historical content changed")
        history_checked += 1

archived = Path("docs/archive/supplements-before-2026-10-14.md").read_text(encoding="utf-8").split("\n\n", 1)[1]
if archived.rstrip() != original(Path("docs/supplements.md")).rstrip():
    errors.append("supplements archive differs from original")

changed = subprocess.check_output(["git", "diff", "--name-only", "-z", "HEAD"]).decode("utf-8").split("\0")
untracked = subprocess.check_output(["git", "ls-files", "--others", "--exclude-standard", "-z"]).decode("utf-8").split("\0")
for name in filter(None, changed + untracked):
    if name != "README.md" and not (name.startswith("docs/") and name.endswith(".md")):
        errors.append("non-document change: " + name)

print(json.dumps({"documents": len(docs), "relative_links": links,
                  "historical_files": history_checked,
                  "line_counts": {p.as_posix(): len(p.read_text(encoding="utf-8").splitlines()) for p in current},
                  "errors": errors}, ensure_ascii=False, indent=2))
raise SystemExit(bool(errors))
```
